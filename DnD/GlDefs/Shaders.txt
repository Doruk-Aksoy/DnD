// Righteous Fire, on the aura model skin.
//
// No type keyword, but the type is NOT what matters here: CheckForTexture defaults to
// TEXMAN_TryAny, so even "Texture" would fall back past a usetype mismatch. What matters is that
// the skin has a NAME at all -- Models/ maps to no namespace, so a lump under it is nameless and
// this block silently does nothing. See the Path split in MODELDEF.aura.
//
// Speed is the one knob this side has: timer arrives at the shader already multiplied by it, so
// every drift, swirl and flicker rate in the shader scales together off this number.
HardwareShader RGHTFIRE {
	Shader "GLDefs/Shaders/RighteousFire.fp"
	Speed 1.0
}

// Annihilus. Same naming rule as RGHTFIRE: the skin lives in Graphics/ so the lump HAS a name,
// which is the only reason this block can find it at all.
HardwareShader ANNIHILS {
	Shader "GLDefs/Shaders/Annihilus.fp"
	Speed 1.0
}

// Anger. Same naming rule as the rest: the skin lives under Graphics/ so the lump HAS a name.
HardwareShader ANGERAUR {
	Shader "GLDefs/Shaders/Anger.fp"
	Speed 1.0
}

// Freezing Pulse. Same naming rule again. Note this one is bound to a MODEL skin rather than a
// sprite, which works only because the lump is under Graphics/ -- a skin loaded from Models/ is
// created with a blank name and no HardwareShader block can ever match it.
HardwareShader FREEZPUL {
	Shader "GLDefs/Shaders/FreezingPulse.fp"
	Speed 1.0
}

// Glacial Spike. Skin under Graphics/ for the same naming reason as the rest.
HardwareShader GLACIAL {

	Speed 1.0
}

// Creeping Frost. Its own texture lump purely so it gets its own shader -- a HardwareShader binds

HardwareShader CRPFROST {
	Shader "GLDefs/Shaders/CreepingFrost.fp"
	Speed 1.0
}
