Actor Orbiter_Moon
{
  Height 32
  Radius 32
  mass 99999
  +FORCEYBILLBOARD
  var int user_count;
  var int user_stock;
  var int user_stockb;
  RenderStyle Add
  Alpha 1
  projectile
  +RIPPER
  +NOTARGETSWITCH
  +NOGRAVITY
  +NOCLIP +NOBLOCKMAP
  DONTHURTSHOOTER
		States{
Spawn:
	RTUR A 0 Bright NoDelay A_SetUserVar("user_stockb",  1575)
Idle:
    MZZN Z  1 Bright A_Warp(AAPTR_TARGET, 561, 0, 100, user_stock, WARPF_ABSOLUTEANGLE|WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE)
	loop
Spawnloop:
	MOON AAA 0 A_SpawnItemEx("RuneFx2",random(-20,20),random(-20,20),random(-60,80),0,0,0)
	RTUR A 0 A_jumpif(user_stockb <=0,"PhaseIncrease")
	RTUR A 0 A_SetUserVar("user_stock",  user_stock+27)
	RTUR A 0 A_SetUserVar("user_stockb",  user_stockb-9)
    MZZN Z 1 Bright A_Warp(AAPTR_TARGET, 400, 0, 100, 0, WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE)
	
	//MZZN Z 1 Bright A_Warp(AAPTR_TARGET, 561, 0, 330, user_stock/35, WARPF_ABSOLUTEANGLE|WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE)
    Loop
PhaseIncrease:
	MZZN Z 0 bright ACS_namedexecutealways("ProcessData",0,user_count,0)
	RTUR A 0 A_SetUserVar("user_count",  user_count+1)
	RTUR A 0 A_SetUserVar("user_stockb",  1575)
	RTUR A 0 ACS_namedexecutealways("moon cycle",0,user_count)
	goto Spawnloop
/*
PhaseIncreaseWait:

	//loop
PhaseIncreaseEnd:
	MZZN Z 1 Bright A_Warp(AAPTR_TARGET, 561, 0, 100, user_stock/35, WARPF_ABSOLUTEANGLE|WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE)
	RTUR A 0 A_SetUserVar("user_count",  user_count+1)
	RTUR A 0 A_SetUserVar("user_stockb",  1575)
	RTUR A 0 ACS_namedexecutealways("moon cycle",0,user_count)
	goto spawnloop*/
Stoploop:
    MZZN Z  1
    Loop
Death:

	RTUR C 4

	loop
}}
